  var coldStartGraceUntil = Date.now() + 2500; // 콜드스타트 직후 2.5초간은 값이 안 안정된 상태이므로 무조건 재적용
  function setAppViewportHeight(){
    var container = document.getElementById('appContainer');
    if(!container) return;
    var vp = window.visualViewport;
    var h = vp ? vp.height : window.innerHeight;
    var ratio = window.innerWidth >= 500 ? 0.92 : 1;
    var newHeight = Math.round(h * ratio);
    var prev = parseInt(container.style.height, 10) || 0;
    var inGracePeriod = !prev || Date.now() < coldStartGraceUntil;
    if (!inGracePeriod && Math.abs(newHeight - prev) < 20) return;
    container.style.height = newHeight + 'px';
  }

  var vphRaf = null;
  function scheduleSetAppViewportHeight(){
    if (vphRaf) cancelAnimationFrame(vphRaf);
    vphRaf = requestAnimationFrame(function(){ vphRaf = null; setAppViewportHeight(); });
  }
  window.addEventListener('resize', scheduleSetAppViewportHeight);
  window.addEventListener('orientationchange', scheduleSetAppViewportHeight);
  if (window.visualViewport){
    window.visualViewport.addEventListener('resize', scheduleSetAppViewportHeight);
  }
  document.addEventListener('DOMContentLoaded', setAppViewportHeight);
  window.addEventListener('load', function(){
    setAppViewportHeight();
    // iOS WKWebView는 콜드스타트 직후 visualViewport 값이 곧바로 안정화되지 않아
    // 첫 측정이 부정확할 수 있음(하단 네비가 화면 밖으로 밀려 안 보이던 원인) -> 잠깐 몇 번 더 재확인
    [100, 400, 900, 1800].forEach(function(delay){ setTimeout(setAppViewportHeight, delay); });
  });
  document.addEventListener('visibilitychange', function(){
    if (!document.hidden) setAppViewportHeight();
  });
  // 콜드스타트 직후 값이 늦게 잡히는 경우 대비해 잠깐 뒤에 한 번 더 재계산
  setTimeout(setAppViewportHeight, 50);
  setTimeout(setAppViewportHeight, 300);
