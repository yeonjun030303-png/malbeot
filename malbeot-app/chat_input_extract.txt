    openFullScreen('profileDetailScreen');

  socket.emit('users:get_one', {userId}, (res)=>{
    fetchedUser = res && res.user; userLoaded = true; tryOpenProfile();

    roomsLoaded = true; tryOpenProfile();
  });
}
